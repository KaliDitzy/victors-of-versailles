sphere_of_influence = {
	imminent_dissolution_of_soi = {
		activation = {
			has_country_flag = has_soi
			is_major = no
		}
		cancel_trigger = {
			OR = {
				NOT = { has_country_flag = has_soi }
				is_major = yes
			}
		}
		available = {
			hidden_trigger = { always = no }
		}
		is_good = no
		days_mission_timeout = 180
		timeout_effect = {
			clr_country_flag = has_soi
		}
	}
	compose_a_sphere_of_influence = {
		visible = {
			NOT = { has_country_flag = has_soi }
			NOT = { has_country_flag = composing_soi }
			is_major = yes
		}
		available = {
			OR = {
				any_country = {
					is_subject_of = PREV
				}
				any_country = {
					is_major = no
					is_subject = no
					OR = {
						is_in_faction_with = PREV
						has_opinion = {
							target = PREV
							value > 79
						}
					}
					check_variable = { current_soi = 0 }
				}
			}
		}
		cost = 150
		complete_effect = {
			set_country_flag = composing_soi
		}
	}
	composing_soi = {
		activation = {
			has_country_flag = composing_soi
		}
		cancel_trigger = {
			OR = {
				has_country_flag = has_soi
				is_major = no
			}
		}
		available = {
			hidden_trigger = { always = no }
		}
		is_good = yes
		days_mission_timeout = 180
		timeout_effect = {
			clr_country_flag = composing_soi
			set_country_flag = has_soi
			every_country = {
				limit = {
					check_variable = { will_join_soi = PREV }
				}
				set_variable = { current_soi = PREV }
				set_variable = { will_join_soi = 0 }
			}
		}
	}
	init_soi_member = {
		target_trigger = {
			FROM = {
				is_major = no
				is_subject = no
				OR = {
					is_in_faction_with = PREV
					has_opinion = {
						target = PREV
						value > 79
					}
				}
				check_variable = { current_soi = 0 }
				check_variable = { will_join_soi = 0 }
			}
		}
		visible = {
			has_country_flag = composing_soi
		}
		cost = 50
		modifier = {
			political_power_factor = -0.05
		}
		days_remove = 30
		remove_effect = {
			if = {
				limit = {
					has_country_flag = composing_soi
				}
				FROM = {
					set_variable = { will_join_soi = PREV }
				}
			}
			else = {
				FROM = {
					set_variable = { current_soi = PREV }
				}
			}
		}
	}
}

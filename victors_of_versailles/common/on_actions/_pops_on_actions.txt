on_actions = {
	on_startup = {
		effect = {
			every_country = {
				set_temp_variable = { pp_d = party_popularity_100@democratic }
				set_temp_variable = { pp_c = party_popularity_100@communism }
				set_temp_variable = { pp_f = party_popularity_100@fascism }
				set_temp_variable = { pp_n = party_popularity_100@neutrality }
				every_controlled_state = {
					limit = {
						NOT = { has_state_flag = defined_pops }
					}
					set_variable_to_random = {
						var = militancy
						min = 0
						max = 0.025
					}
					set_variable_to_random = {
						var = consciousness
						min = 0
						max = 0.05
					}
					set_variable = { democratic_support = pp_d }
					set_variable = { communism_support = pp_c }
					set_variable = { fascism_support = pp_f }
					set_variable = { neutrality_support = pp_n }
				}
				calculate_party_popularity = yes
				
				set_temp_variable = { amt_states = 0 }
				set_temp_variable = { total_militancy = 0 }
				set_temp_variable = { total_consciousness = 0 }
				set_temp_variable = { current_country = THIS }
				every_controlled_state = {
					add_to_temp_variable = { amt_states = 1 }

					add_to_temp_variable = { total_militancy = militancy }
					add_to_temp_variable = { total_consciousness = consciousness }
				}
				set_variable = { militancy = total_militancy }
				divide_variable = { militancy = amt_states }
				set_variable = { consciousness = total_consciousness }
				divide_variable = { consciousness = amt_states }
				#

				calculate_unity_penalty_from_pops = yes

				add_dynamic_modifier = { modifier = pop_modifiers }
			}
		}
	}
	on_daily = {
		effect = {
			if = {
				limit = {
					NOT = {
						has_dynamic_modifier = {
							modifier = pop_modifiers
						}
					}
				}
				add_dynamic_modifier = {
					modifier = pop_modifiers
				}
			}
		}
		effect = {
			calculate_party_popularity = yes
		}
		effect = {  # For more important, immediate changes in pops
			set_temp_variable = { amt_states = 0 }
			set_temp_variable = { total_militancy = 0 }
			set_temp_variable = { total_consciousness = 0 }
			set_temp_variable = { current_country = THIS }
			every_controlled_state = {
				add_to_temp_variable = { amt_states = 1 }

				#
				set_temp_variable = { militancy_add = 0 }
				set_temp_variable = { militancy_factor = 1 }
				set_temp_variable = { consciousness_add = 0 }
				set_temp_variable = { consciousness_factor = 1 }
				###
				
				if = {
					limit = {
						any_neighbor_state = {
							controller = {
								divisions_in_state = {
									size > 0
									state = PREV
								}
								has_war = yes
								NOT = { tag = var:current_country }
							}
						}
					}
					add_to_temp_variable = { consciousness_add = 0.002 }
					add_to_temp_variable = { militancy_factor = 0.05 }
				}

				if = {
					limit = {
						var:current_country = {
							has_war = yes
						}
					}
					add_to_temp_variable = { militancy_add = 0.001 }
				}

				if = {
					limit = {
						any_country = {
							var:current_country = {
								is_justifying_wargoal_against = PREV
							}
						}
					}
					add_to_temp_variable = { consciousness_add = 0.001 }
				}

				set_temp_variable = { militancy_from_consciousness = consciousness }
				subtract_from_temp_variable = { militancy_from_consciousness = 0.3 }
				divide_temp_variable = { militancy_from_consciousness = 150 }
				add_to_temp_variable = { militancy_add = militancy_from_consciousness }

				## PARTY DRIFT
				if = {
					limit = {
						check_variable = { democratic_support > 0 }
					}
					set_temp_variable_to_random = emigrate_rng
					set_temp_variable = { pass_democratic_support = democratic_support }
					divide_temp_variable = { pass_democratic_support = 560 }
					random_neighbor_state = {
						limit = {
							OR = {
								NOT = { check_variable = { emigrate_rng < 0.7 } }
								owner = {
									tag = prev.prev.owner
								}
							}
						}
						PREV = {
							subtract_from_variable = { democratic_support = pass_democratic_support }
						}
						add_to_variable = { democratic_support = pass_democratic_support }
					}
				}
				if = {
					limit = {
						check_variable = { communism_support > 0 }
					}
					set_temp_variable_to_random = emigrate_rng
					set_temp_variable = { pass_communism_support = communism_support }
					divide_temp_variable = { pass_communism_support = 560 }
					random_neighbor_state = {
						limit = {
							OR = {
								NOT = { check_variable = { emigrate_rng < 0.7 } }
								owner = {
									tag = prev.prev.owner
								}
							}
						}
						PREV = {
							subtract_from_variable = { communism_support = pass_communism_support }
						}
						add_to_variable = { communism_support = pass_communism_support }
					}
				}
				if = {
					limit = {
						check_variable = { fascism_support > 0 }
					}
					set_temp_variable_to_random = emigrate_rng
					set_temp_variable = { pass_fascism_support = fascism_support }
					divide_temp_variable = { pass_fascism_support = 560 }
					random_neighbor_state = {
						limit = {
							OR = {
								NOT = { check_variable = { emigrate_rng < 0.7 } }
								owner = {
									tag = prev.prev.owner
								}
							}
						}
						PREV = {
							subtract_from_variable = { fascism_support = pass_fascism_support }
						}
						add_to_variable = { fascism_support = pass_fascism_support }
					}
				}
				if = {
					limit = {
						check_variable = { neutrality_support > 0 }
					}
					set_temp_variable_to_random = emigrate_rng
					set_temp_variable = { pass_neutrality_support = neutrality_support }
					divide_temp_variable = { pass_neutrality_support = 560 }
					random_neighbor_state = {
						limit = {
							OR = {
								NOT = { check_variable = { emigrate_rng < 0.7 } }
								owner = {
									tag = prev.prev.owner
								}
							}
						}
						PREV = {
							subtract_from_variable = { neutrality_support = pass_neutrality_support }
						}
						add_to_variable = { neutrality_support = pass_neutrality_support }
					}
				}

				## MILITANCY/CONSCIOUSNESS DRIFT
				if = {
					limit = {
						check_variable = { militancy > 0 }
					}
					set_temp_variable_to_random = emigrate_rng
					set_temp_variable = { pass_militancy = militancy }
					divide_temp_variable = { pass_militancy = 70 }
					random_neighbor_state = {
						limit = {
							OR = {
								NOT = { check_variable = { emigrate_rng < 0.35 } }
								owner = {
									tag = prev.prev.owner
								}
							}
						}
						PREV = {
							subtract_from_variable = { militancy = pass_militancy }
						}
						add_to_variable = { militancy = pass_militancy }
					}
				}

				if = {
					limit = {
						check_variable = { consciousness > 0 }
					}
					#set_temp_variable_to_random = emigrate_rng
					set_temp_variable = { pass_consciousness = consciousness }
					divide_temp_variable = { pass_consciousness = 70 }
					random_neighbor_state = {
						PREV = {
							subtract_from_variable = { consciousness = pass_consciousness }
						}
						add_to_variable = { consciousness = pass_consciousness }
					}
				}

				###
				multiply_temp_variable = { militancy_add = militancy_factor }
				multiply_temp_variable = { consciousness_add = consciousness_factor }
				add_to_variable = { militancy = militancy_add }
				add_to_variable = { consciousness = consciousness_add }
				clamp_variable = {
					var = militancy
					min = 0
					max = 1 
				}
				clamp_variable = {
					var = consciousness
					min = 0
					max = 1 
				}
				#

				add_to_temp_variable = { total_militancy = militancy }
				add_to_temp_variable = { total_consciousness = consciousness }
			}
			set_variable = { militancy = total_militancy }
			divide_variable = { militancy = amt_states }
			set_variable = { consciousness = total_consciousness }
			divide_variable = { consciousness = amt_states }

			calculate_unity_penalty_from_pops = yes
		}
	}
	on_weekly = { # For less important, gradual changes in pops
		effect = {
			set_temp_variable = { amt_states = 0 }
			set_temp_variable = { total_militancy = 0 }
			set_temp_variable = { total_consciousness = 0 }
			set_temp_variable = { current_country = THIS }
			every_controlled_state = {
				add_to_temp_variable = { amt_states = 1 }

				#
				set_temp_variable = { militancy_add = 0 }
				set_temp_variable = { militancy_factor = 1 }
				set_temp_variable = { consciousness_add = 0 }
				set_temp_variable = { consciousness_factor = 1 }
				###

				if = {
					limit = {
						any_neighbor_state = {
							check_variable = { militancy > 0.9 }
						}
					}
					add_resistance = 0.25
				}
				if = {
					limit = {
						any_neighbor_state = {
							check_variable = { militancy > 0.8 }
						}
					}
					add_resistance = 0.25
				}
				if = {
					limit = {
						any_neighbor_state = {
							check_variable = { militancy > 0.7 }
						}
					}
					add_resistance = 0.25
				}
				if = {
					limit = {
						any_neighbor_state = {
							check_variable = { militancy > 0.6 }
						}
					}
					add_resistance = 0.25
				}
				if = {
					limit = {
						any_neighbor_state = {
							check_variable = { militancy > 0.5 }
						}
					}
					add_resistance = 0.25
				}
				if = {
					limit = {
						any_neighbor_state = {
							check_variable = { militancy > 0.4 }
						}
					}
					add_resistance = 0.25
				}
				if = {
					limit = {
						any_neighbor_state = {
							check_variable = { militancy > 0.3 }
						}
					}
					add_resistance = 0.25
				}
				if = {
					limit = {
						any_neighbor_state = {
							check_variable = { militancy > 0.2 }
						}
					}
					add_resistance = 0.25
				}
				if = {
					limit = {
						any_neighbor_state = {
							check_variable = { militancy > 0.1 }
						}
					}
					add_resistance = 0.25
				}

				###
				multiply_temp_variable = { militancy_add = militancy_factor }
				multiply_temp_variable = { consciousness_add = consciousness_factor }
				add_to_variable = { militancy = militancy_add }
				add_to_variable = { consciousness = consciousness_add }
				clamp_variable = {
					var = militancy
					min = 0
					max = 1 
				}
				clamp_variable = {
					var = consciousness
					min = 0
					max = 1 
				}
				#

				add_to_temp_variable = { total_militancy = militancy }
				add_to_temp_variable = { total_consciousness = consciousness }
			}
			set_variable = { militancy = total_militancy }
			divide_variable = { militancy = amt_states }
			set_variable = { consciousness = total_consciousness }
			divide_variable = { consciousness = amt_states }

			calculate_unity_penalty_from_pops = yes
		}
	}
}
